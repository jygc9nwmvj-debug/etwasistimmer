  <script>
    if (typeof gsap !== 'undefined' && typeof ScrollTrigger !== 'undefined') {
      gsap.registerPlugin(ScrollTrigger);

      gsap.set("#contentStage", { opacity: 0, y: 30 });
      gsap.set("#topNav", { opacity: 0 });

      let cachedX = 0;
      let cachedY = 0;

      function calculateTarget() {
        const hero = document.getElementById('heroTitle');
        if (!hero) return;

        gsap.set(hero, { x: 0, y: 0 });
        const rect = hero.getBoundingClientRect();
        const isMobile = window.innerWidth <= 768;
        
        // Exakte Zielkoordinaten oben links (Bündig mit 5vw Padding)
        const targetTop = isMobile ? window.innerWidth * 0.06 : 48;
        const targetLeft = window.innerWidth * (isMobile ? 0.06 : 0.05);

        cachedX = isMobile ? 0 : (targetLeft - rect.left);
        cachedY = targetTop - rect.top;
      }

      calculateTarget();

      const tl = gsap.timeline({
        scrollTrigger: {
          trigger: ".scroll-container",
          start: "top top",
          end: "100% top",
          scrub: 0.5,
          snap: {
            snapTo: 1,
            duration: { min: 0.3, max: 0.6 },
            delay: 0.1,
            ease: "power1.inOut"
          }
        }
      });

      // 1. Gedichtzeilen ausblenden
      tl.to(".fade-line", {
        opacity: 0,
        y: -15,
        duration: 0.3
      }, 0);

      // 2. "Etwas ist immer." wandert exakt an die freie Position oben links
      tl.to("#heroTitle", {
        x: () => cachedX,
        y: () => cachedY,
        duration: 0.8,
        ease: "power2.out"
      }, 0);

      // 3. Background Color-Flip (Dark zu Light)
      tl.to("body", {
        backgroundColor: "#f4f3ef",
        color: "#121211",
        duration: 0.6
      }, 0.2);

      tl.to(["#nameSub", ".col-card h2", ".card-desc", ".top-nav a"], {
        color: "#706f6b",
        duration: 0.6
      }, 0.2);

      // 4. Content & Navigation faden ein
      tl.to(["#contentStage", "#topNav"], {
        opacity: 1,
        y: 0,
        duration: 0.5
      }, 0.4);

      document.getElementById("heroTitle").addEventListener("click", () => {
        window.scrollTo({ top: 0, behavior: 'smooth' });
      });

      let resizeTimer;
      window.addEventListener('resize', () => {
        clearTimeout(resizeTimer);
        resizeTimer = setTimeout(() => {
          calculateTarget();
          ScrollTrigger.refresh();
        }, 150);
      });
    }
  </script>
